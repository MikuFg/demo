-- Создание базы данных
CREATE DATABASE IF NOT EXISTS shoes2;
USE shoes2;

-- Создание таблиц
CREATE TABLE IF NOT EXISTS `manufactures` (
  `id_manufacture` int NOT NULL AUTO_INCREMENT,
  `manufacture_name` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_manufacture`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `order_station` (
  `id_station` int NOT NULL AUTO_INCREMENT,
  `indexes` int DEFAULT NULL,
  `city` varchar(50) DEFAULT NULL,
  `street` varchar(50) DEFAULT NULL,
  `house` int DEFAULT NULL,
  PRIMARY KEY (`id_station`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `order_statuses` (
  `id_status` int NOT NULL AUTO_INCREMENT,
  `status_name` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `roles` (
  `id_role` int NOT NULL AUTO_INCREMENT,
  `role_name` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_role`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `users` (
  `id_user` int NOT NULL AUTO_INCREMENT,
  `last_name` varchar(50) DEFAULT NULL,
  `first_name` varchar(50) DEFAULT NULL,
  `family_name` varchar(50) DEFAULT NULL,
  `user_role_id` int DEFAULT NULL,
  `login` varchar(50) DEFAULT NULL,
  `passw` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_user`),
  KEY `user_role_id` (`user_role_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `product_category` (
  `id_product_category` int NOT NULL AUTO_INCREMENT,
  `category_name` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_product_category`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `product_form` (
  `id_product_form` int NOT NULL AUTO_INCREMENT,
  `product_form_name` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_product_form`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `providers` (
  `id_provider` int NOT NULL AUTO_INCREMENT,
  `provider_name` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_provider`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `product` (
  `article` varchar(50) NOT NULL,
  `product_form_id` int DEFAULT NULL,
  `unit` varchar(50) DEFAULT NULL,
  `price` float DEFAULT NULL,
  `provider_id` int DEFAULT NULL,
  `manufacture_id` int DEFAULT NULL,
  `category_id` int DEFAULT NULL,
  `current_discount` int DEFAULT NULL,
  `quantity` int DEFAULT NULL,
  `product_description` varchar(500) DEFAULT NULL,
  `image` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`article`),
  KEY `product_form_id` (`product_form_id`),
  KEY `provider_id` (`provider_id`),
  KEY `manufacture_id` (`manufacture_id`),
  KEY `category_id` (`category_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `orders` (
  `id_order` int NOT NULL AUTO_INCREMENT,
  `order_date` date DEFAULT NULL,
  `delivery_date` date DEFAULT NULL,
  `order_station_id` int DEFAULT NULL,
  `client_id` int DEFAULT NULL,
  `code_for_get` int DEFAULT NULL,
  `status_id` int DEFAULT NULL,
  PRIMARY KEY (`id_order`),
  KEY `order_station_id` (`order_station_id`),
  KEY `client_id` (`client_id`),
  KEY `status_id` (`status_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `orders_products` (
  `id_orders_products` int NOT NULL AUTO_INCREMENT,
  `order_id` int DEFAULT NULL,
  `product_id` varchar(50) DEFAULT NULL,
  `quantity` int DEFAULT NULL,
  PRIMARY KEY (`id_orders_products`),
  KEY `order_id` (`order_id`),
  KEY `product_id` (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание представления
CREATE OR REPLACE VIEW `main_data` AS 
SELECT 
  `p`.`image` AS `image`,
  `p_c`.`category_name` AS `category_name`,
  `p_f`.`product_form_name` AS `product_form_name`,
  `p`.`product_description` AS `product_description`,
  `m`.`manufacture_name` AS `manufacture_name`,
  `pr`.`provider_name` AS `provider_name`,
  `p`.`price` AS `price`,
  `p`.`unit` AS `unit`,
  `p`.`quantity` AS `quantity`,
  `p`.`current_discount` AS `current_discount`,
  `p`.`article` AS `article` 
FROM ((((`product` `p` 
  JOIN `product_category` `p_c` ON (`p_c`.`id_product_category` = `p`.`category_id`)) 
  JOIN `product_form` `p_f` ON (`p_f`.`id_product_form` = `p`.`product_form_id`)) 
  JOIN `manufactures` `m` ON (`m`.`id_manufacture` = `p`.`manufacture_id`)) 
  JOIN `providers` `pr` ON (`pr`.`id_provider` = `p`.`provider_id`));
